+ camera.position.set(Math.sin(yaw)*Math.cos(pitch),Math.sin(pitch),Math.cos(yaw)*Math.cos(pitch)).multiplyScalar(p.distance).add(target);

+ camera.fov=p.fov;camera.aspect=width/height;
